<script>
  // ═══════════════════════════════════════════════════════════
  // 🔐 API Configuration
  // ═══════════════════════════════════════════════════════════
  const API = "https://script.google.com/macros/s/AKfycbxojGj0LzZFLFFnUAO0A2wHYjaM1GewPPWhwKJ2eutwm3G4pM0tJwlMI8fk0-jJKgmrpQ/exec";
  
  // ═══════════════════════════════════════════════════════════
  // 🔄 Service Worker Registration
  // ═══════════════════════════════════════════════════════════
  window.addEventListener('load', () => {
    // Register Service Worker
    if ('serviceWorker' in navigator) {
      navigator.serviceWorker.register('sw.js')
        .then((reg) => {
          console.log('✅ Service Worker registered:', reg);
          // Update checker
          setInterval(() => {
            reg.update();
          }, 3600000); // Check every hour
        })
        .catch((err) => {
          console.warn('⚠️ Service Worker registration failed:', err);
        });
    }
    
    // PWA Install Banner
    let deferredPrompt;
    window.addEventListener('beforeinstallprompt', (e) => {
      e.preventDefault();
      deferredPrompt = e;
      const banner = document.getElementById('install-banner');
      banner.classList.add('show');
      
      document.getElementById('install-btn').addEventListener('click', () => {
        if (deferredPrompt) {
          deferredPrompt.prompt();
          deferredPrompt.userChoice.then((choiceResult) => {
            if (choiceResult.outcome === 'accepted') {
              console.log('✅ PWA installed');
              banner.classList.remove('show');
            }
            deferredPrompt = null;
          });
        }
      });
      
      document.getElementById('dismiss-btn').addEventListener('click', () => {
        banner.classList.remove('show');
        deferredPrompt = null;
      });
    });
    
    // Handle app installed
    window.addEventListener('appinstalled', () => {
      console.log('🎉 App was successfully installed!');
      document.getElementById('install-banner').classList.remove('show');
    });
  });
  
  // ═══════════════════════════════════════════════════════════
  // 💾 Global State
  // ═══════════════════════════════════════════════════════════
  const ST = {
    user: null,
    sows: [],
    boars: [],
    settings: {},
    filterStatus: "ALL",
    cullSowId: null,
    isOnline: navigator.onLine
  };
  
  // Online/Offline indicator
  window.addEventListener('online', () => {
    ST.isOnline = true;
    document.getElementById('conn-dot').classList.remove('offline');
    showToast('✅ เชื่อมต่ออินเทอร์เน็ตแล้ว', 'success');
  });
  
  window.addEventListener('offline', () => {
    ST.isOnline = false;
    document.getElementById('conn-dot').classList.add('offline');
    showToast('⚠️ ไม่มีการเชื่อมต่ออินเทอร์เน็ต', 'warning');
  });
  
  // ═══════════════════════════════════════════════════════════
  // 🚀 API Caller (unchanged from original)
  // ═══════════════════════════════════════════════════════════
  async function callAPI(action, payload = {}) {
    const url = API + '?action=' + encodeURIComponent(action) + '&payload=' + encodeURIComponent(JSON.stringify(payload));
    try {
      const res = await fetch(url, { method: 'GET', redirect: 'follow', cache: 'no-store' });
      const data = JSON.parse(await res.text());
      return data;
    } catch (err) {
      console.error('❌ API Error:', err);
      showToast('❌ เชื่อมต่อไม่ได้', 'error');
      return { success: false, error: err.message };
    }
  }
  
  // Test connection on startup
  window.addEventListener('load', () => {
    callAPI('ping').then((res) => {
      const banner = document.getElementById('conn-banner');
      if (res.success) {
        banner.innerHTML = '✅ เชื่อมต่อสำเร็จ';
        banner.className = 'conn-banner success';
      } else {
        banner.innerHTML = '❌ เชื่อมต่อไม่ได้ — ตรวจ API URL';
        banner.className = 'conn-banner error';
      }
    });
  });
  
  // ═══════════════════════════════════════════════════════════
  // 🎨 UI Functions (rest remains the same)
  // ═══════════════════════════════════════════════════════════
  
  function showPage(pageId) {
    document.querySelectorAll('.page').forEach(p => p.classList.remove('active'));
    document.getElementById(pageId).classList.add('active');
    document.querySelectorAll('.nav-btn').forEach(b => b.classList.remove('active'));
    event.target.closest('.nav-btn')?.classList.add('active');
  }
  
  function showToast(msg, type = 'info') {
    const toast = document.createElement('div');
    toast.className = 'toast toast-' + type;
    toast.textContent = msg;
    document.getElementById('toasts').appendChild(toast);
    setTimeout(() => toast.remove(), 3000);
  }
  
  function openModal(modalId) {
    document.getElementById(modalId).style.display = 'flex';
  }
  
  function closeModal(modalId) {
    document.getElementById(modalId).style.display = 'none';
  }
  
  // Login functions (unchanged)
  async function doLogin(username, password, role, display_name) {
    showLoader('กำลังเข้าสู่ระบบ…');
    const res = await callAPI('login', { username, password });
    hideLoader();
    if (res.success) {
      ST.user = res.user || { username, role, display_name };
      loadDashboard();
      document.getElementById('screen-login').classList.remove('active');
      document.getElementById('screen-app').classList.add('active');
      document.getElementById('uname').textContent = ST.user.display_name;
    } else {
      showToast('❌ Login failed', 'error');
    }
  }
  
  function logout() {
    ST.user = null;
    document.getElementById('screen-app').classList.remove('active');
    document.getElementById('screen-login').classList.add('active');
  }
  
  function showLoader(text = 'กำลังโหลด…') {
    document.getElementById('loader-text').textContent = text;
    document.getElementById('ldr').style.display = 'flex';
  }
  
  function hideLoader() {
    document.getElementById('ldr').style.display = 'none';
  }
  
  // Dashboard (stub)
  async function loadDashboard() {
    const res = await callAPI('getDashboard');
    if (res.success) {
      console.log('Dashboard loaded:', res.data);
      // Update UI (implement full dashboard logic)
    }
  }
  
  // Add Sow (stub)
  async function doAddSow() {
    const ear_tag = document.getElementById('new-ear-tag').value;
    const breed = document.getElementById('new-breed').value;
    const res = await callAPI('addSow', { ear_tag, breed });
    if (res.success) {
      showToast('✅ เพิ่มแม่สุกรสำเร็จ', 'success');
      closeModal('m-addsow');
    } else {
      showToast('❌ เพิ่มไม่ได้', 'error');
    }
  }
  
  // Initialize
  console.log('🐷 นิพนธ์ฟาร์ม v2.0 loaded');
</script>
